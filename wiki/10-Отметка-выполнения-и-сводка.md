# 1. Отметка выполнения (55–70 мин)

Добавьте в `ITaskService`:

```csharp
Task<TaskDto> ToggleCompletedAsync(long userId, long id, CancellationToken cancellationToken);
```

`src/api/TaskPlanner.Api/Services/TaskService.Complete.cs`:

```csharp
using Microsoft.EntityFrameworkCore;
using TaskPlanner.Api.Mappers;
using TaskPlanner.Core.Dtos;
using TaskPlanner.Core.Enums;
using TaskPlanner.Core.Exceptions;

namespace TaskPlanner.Api.Services;

public partial class TaskService
{
    /// <summary>
    /// Переключает задачу: была выполнена — вернуть в работу,
    /// была невыполненной — отметить выполненной.
    /// </summary>
    public async Task<TaskDto> ToggleCompletedAsync(
        long userId,
        long id,
        CancellationToken cancellationToken)
    {
        var task = await db.Tasks
            .FirstOrDefaultAsync(x => x.Id == id && x.UserId == userId, cancellationToken);

        if (task is null)
        {
            throw ApiException.NotFound("Задача не найдена");
        }

        var now = DateTimeOffset.UtcNow;

        if (task.Status == TaskStatus.Done)
        {
            // Возврат в работу: срок выполнения обязательно сбрасывается,
            // иначе CHECK ck_tasks_completed_at не пройдёт.
            task.Status = TaskStatus.New;
            task.CompletedAt = null;
        }
        else
        {
            task.Status = TaskStatus.Done;
            task.CompletedAt = now;
        }

        task.UpdatedAt = now;

        await db.SaveChangesAsync(cancellationToken);

        logger.LogInformation(
            "Пользователь {UserId} переключил задачу {TaskId}: статус {Status}",
            userId, id, task.Status);

        return task.ToDto();
    }
}
```

Метод в контроллере:

```csharp
/// <summary>Отметить задачу выполненной или вернуть её в работу.</summary>
[HttpPatch("{id:long}/complete")]
[ProducesResponseType(typeof(ApiResponse<TaskDto>), StatusCodes.Status200OK)]
[ProducesResponseType(typeof(ApiResponse<object?>), StatusCodes.Status404NotFound)]
public async Task<IActionResult> ToggleCompleted(long id, CancellationToken cancellationToken)
{
    var task = await taskService.ToggleCompletedAsync(
        currentUser.GetUserId(), id, cancellationToken);

    var message = task.Status == Enums.TaskStatus.Done
        ? "Задача отмечена выполненной"
        : "Задача возвращена в работу";

    return Ok(ApiResponse<TaskDto>.Ok(task, message));
}
```

И добавьте в начало файла контроллера:

```csharp
using TaskPlanner.Core.Enums;
```

> **Замечание:** `PATCH`, а не `PUT`, потому что меняется одно состояние, а не
> весь объект. Клиенту достаточно прислать факт «переключи», тело запроса
> не нужно.

# 2. Сводка через SQL (70–90 мин)

Сводка — пять чисел. Считать их в приложении нельзя: при тысяче задач это
лишняя передача данных и нагрузка на память. Считаем в базе.

`src/api/TaskPlanner.Api/Services/ISummaryService.cs`:

```csharp
using TaskPlanner.Core.Dtos;

public interface ISummaryService
{
    Task<TaskSummaryDto> GetAsync(long userId, CancellationToken cancellationToken);
}
```

`src/api/TaskPlanner.Core/Dtos/TaskSummaryDto.cs`:

```csharp
namespace TaskPlanner.Core.Dtos;

public class TaskSummaryDto
{
    public long Total { get; set; }

    public long Done { get; set; }

    public long NotDone { get; set; }

    public long Overdue { get; set; }

    /// <summary>Незавершённые задачи по приоритетам: low, medium, high.</summary>
    public Dictionary<string, long> ByPriority { get; set; } = new();
}
```

`src/api/TaskPlanner.Api/Services/SummaryService.cs`:

```csharp
using System.Data.Common;
using Microsoft.EntityFrameworkCore;
using TaskPlanner.Api.Data;
using TaskPlanner.Core.Dtos;

namespace TaskPlanner.Api.Services;

public class SummaryService(AppDbContext db) : ISummaryService
{
    public async Task<TaskSummaryDto> GetAsync(
        long userId,
        CancellationToken cancellationToken)
    {
        // Database.SqlQuery<T> выполняет SELECT и мапит результат по ИМЕНАМ КОЛОНОК.
        // Имена ровно такие, как в алиасах: Total, Done, NotDone, Overdue,
        // OpenLow, OpenMedium, OpenHigh. Регистр учитывается.
        var rows = await db.Database
            .SqlQuery<SummaryRow>($"""
                SELECT
                    COUNT(*)                                          AS "Total",
                    COUNT(*) FILTER (WHERE status = 'done')           AS "Done",
                    COUNT(*) FILTER (WHERE status <> 'done')          AS "NotDone",
                    COUNT(*) FILTER (
                        WHERE status <> 'done'
                          AND due_date IS NOT NULL
                          AND due_date < CURRENT_DATE)              AS "Overdue",
                    COUNT(*) FILTER (
                        WHERE status <> 'done' AND priority = 'low')    AS "OpenLow",
                    COUNT(*) FILTER (
                        WHERE status <> 'done' AND priority = 'medium') AS "OpenMedium",
                    COUNT(*) FILTER (
                        WHERE status <> 'done' AND priority = 'high')   AS "OpenHigh"
                FROM tasks
                WHERE user_id = {userId}
                """)
            .ToListAsync(cancellationToken);

        var row = rows.Single();

        return new TaskSummaryDto
        {
            Total = row.Total,
            Done = row.Done,
            NotDone = row.NotDone,
            Overdue = row.Overdue,

            // byPriority в контракте содержит только незавершённые:
            // выполненная задача уже не требует внимания.
            ByPriority = new Dictionary<string, long>
            {
                ["low"] = row.OpenLow,
                ["medium"] = row.OpenMedium,
                ["high"] = row.OpenHigh,
            },
        };
    }

    /// <summary>
    /// Тип для маппинга результата сырого SELECT.
    /// Свойства должны совпадать с алиасами в запросе.
    /// </summary>
    private sealed class SummaryRow
    {
        public long Total { get; set; }
        public long Done { get; set; }
        public long NotDone { get; set; }
        public long Overdue { get; set; }
        public long OpenLow { get; set; }
        public long OpenMedium { get; set; }
        public long OpenHigh { get; set; }
    }
}
```

Регистрация и метод контроллера:

```csharp
// Program.cs
builder.Services.AddScoped<ISummaryService, SummaryService>();
```

```csharp
// TasksController
/// <summary>Сводка: всего, выполнено, не выполнено, просрочено.</summary>
[HttpGet("summary")]
[ProducesResponseType(typeof(ApiResponse<TaskSummaryDto>), StatusCodes.Status200OK)]
public async Task<IActionResult> GetSummary(CancellationToken cancellationToken)
{
    var summary = await summaryService.GetAsync(
        currentUser.GetUserId(), cancellationToken);

    return Ok(ApiResponse<TaskSummaryDto>.Ok(summary));
}
```

Добавьте `ISummaryService summaryService` в конструктор контроллера и
`using TaskPlanner.Core.Dtos;`.

> **Замечание:** `userId` подставляется через интерполяцию `$"""...{userId}..."""` —
> это создаёт `DbParameter`, а не склейку строки. Значение уходит в базу
> параметризованным, SQL-инъекция невозможна. **Никогда** не пишите так:
> `$"... WHERE user_id = {userId}"` строковой конкатенацией — это уже дыра.

> **Замечание:** `COUNT(*) FILTER (WHERE ...)` — условный агрегат PostgreSQL
> 9.4+. Он выполняется в базе одним проходом. Альтернатива — пять отдельных
> `COUNT` запросов или загрузка всех строк в память; оба варианта хуже.
> Если очень нужен перенос на другую СУБД, условие переносится в `CASE WHEN
> ... THEN 1 ELSE 0 END` с суммированием — результат тот же.

# Если что-то не получилось

| Симптом | Что делать |
|---|---|
| `Invalid column name 'Total'` при `SqlQuery` | алиас в SQL и имя свойства не совпадают. Проверьте кавычки: `AS "Total"` |
| `No column matches the name` | в `SummaryRow` есть свойство, которого нет в SELECT, или наоборот |
| Сводка считает чужие задачи | в `WHERE user_id = {userId}` потерян параметр |
| `PATCH` возвращает `405` | метод не совпадает с шаблоном маршрута: должно быть `"{id:long}/complete"` |
| `500` про `completed_at` | в `ToggleCompleted` при возврате в работу забыли `CompletedAt = null` |
| `byPriority` пустой | словарь заполняется в коде после запроса — проверьте, что `ByPriority` не переопределён |
| Клиент не видит `byPriority` | сериализация словаря в camelCase не нужна: ключи `low`/`medium`/`high` задаёт сервер |
