# 1. Маршрутизация (40–45 мин)

`src/web/src/App.tsx`:

```tsx
import { BrowserRouter, Route, Routes } from 'react-router-dom';
import { AuthProvider } from './auth/AuthContext';
import { RequireAuth, RequireGuest } from './auth/RequireAuth';
import { LoginPage } from './pages/LoginPage';
import { RegisterPage } from './pages/RegisterPage';
import { TasksPage } from './pages/TasksPage';
import { SummaryPage } from './pages/SummaryPage';

export function App() {
  return (
    <BrowserRouter>
      <AuthProvider>
        <Routes>
          {/* Гость: если уже вошёл — уводим на задачи */}
          <Route path="/login" element={<RequireGuest><LoginPage /></RequireGuest>} />
          <Route path="/register" element={<RequireGuest><RegisterPage /></RequireGuest>} />

          {/* Гость: без токена — на страницу входа */}
          <Route path="/tasks" element={<RequireAuth><TasksPage /></RequireAuth>} />
          <Route path="/summary" element={<RequireAuth><SummaryPage /></RequireAuth>} />

          <Route path="*" element={<RequireAuth><TasksPage /></RequireAuth>} />
        </Routes>
      </AuthProvider>
    </BrowserRouter>
  );
}
```

# 2. Форма входа (45–75 мин)

`src/web/src/pages/LoginPage.tsx`:

```tsx
import { useState, type FormEvent } from 'react';
import { Link, useNavigate } from 'react-router-dom';
import { authApi, errorMessage } from '../api/client';
import { tokenStorage } from '../api/tokenStorage';
import styles from './LoginPage.module.css';

export function LoginPage() {
  const navigate = useNavigate();

  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState<string | null>(null);
  const [submitting, setSubmitting] = useState(false);

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setError(null);

    if (!email.trim() || !password) {
      setError('Заполните email и пароль');
      return;
    }

    setSubmitting(true);

    try {
      const result = await authApi.login({ email: email.trim(), password });

      // Токен в хранилище — иначе перезагрузка страницы выбросит на /login
      tokenStorage.setToken(result.token);
      tokenStorage.setUser(result.user);

      navigate('/tasks', { replace: true });
    } catch (e) {
      setError(errorMessage(e));
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <main className={styles.page}>
      <form className={styles.card} onSubmit={handleSubmit} noValidate>
        <h1 className={styles.title}>Планировщик задач</h1>
        <p className={styles.subtitle}>Войдите, чтобы увидеть свои задачи</p>

        {error && <div className="error" role="alert">{error}</div>}

        <div className="field">
          <label htmlFor="email">Email</label>
          <input
            id="email"
            type="email"
            autoComplete="email"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            placeholder="student@college.ru"
          />
        </div>

        <div className="field">
          <label htmlFor="password">Пароль</label>
          <input
            id="password"
            type="password"
            autoComplete="current-password"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
          />
        </div>

        <button className="btn-primary" type="submit" disabled={submitting}>
          {submitting ? 'Входим…' : 'Войти'}
        </button>

        <p className={styles.hint}>
          Нет аккаунта? <Link to="/register">Зарегистрируйтесь</Link>
        </p>
      </form>
    </main>
  );
}
```

# 3. Форма регистрации (75–95 мин)

`src/web/src/pages/RegisterPage.tsx`:

```tsx
import { useState, type FormEvent } from 'react';
import { Link, useNavigate } from 'react-router-dom';
import { authApi, errorMessage } from '../api/client';
import styles from './LoginPage.module.css';

interface FieldErrors {
  email?: string;
  password?: string;
  fullName?: string;
}

export function RegisterPage() {
  const navigate = useNavigate();

  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [fullName, setFullName] = useState('');

  const [fieldErrors, setFieldErrors] = useState<FieldErrors>({});
  const [error, setError] = useState<string | null>(null);
  const [submitting, setSubmitting] = useState(false);

  function validate(): boolean {
    const errors: FieldErrors = {};

    if (!email.trim()) {
      errors.email = 'Email обязателен';
    } else if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email.trim())) {
      errors.email = 'Некорректный email';
    }

    if (!password) {
      errors.password = 'Пароль обязателен';
    } else if (password.length < 8) {
      errors.password = 'Пароль не короче 8 символов';
    } else if (!/[A-Za-z]/.test(password) || !/[0-9]/.test(password)) {
      errors.password = 'Нужны латинская буква и цифра';
    }

    if (!fullName.trim()) {
      errors.fullName = 'Имя обязательно';
    }

    setFieldErrors(errors);
    return Object.keys(errors).length === 0;
  }

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setError(null);

    if (!validate()) return;

    setSubmitting(true);

    try {
      // Регистрация по контракту токен не возвращает.
      // Сразу делаем вход, чтобы пользователь не вводил данные дважды.
      await authApi.register({
        email: email.trim(),
        password,
        fullName: fullName.trim(),
      });

      const result = await authApi.login({ email: email.trim(), password });

      localStorage.setItem('taskplanner.token', result.token);
      localStorage.setItem('taskplanner.user', JSON.stringify(result.user));

      navigate('/tasks', { replace: true });
    } catch (e) {
      setError(errorMessage(e));
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <main className={styles.page}>
      <form className={styles.card} onSubmit={handleSubmit} noValidate>
        <h1 className={styles.title}>Регистрация</h1>
        <p className={styles.subtitle}>Создайте аккаунт для своих задач</p>

        {error && <div className="error" role="alert">{error}</div>}

        <div className="field">
          <label htmlFor="fullName">Имя и фамилия</label>
          <input
            id="fullName"
            value={fullName}
            onChange={(e) => setFullName(e.target.value)}
            autoComplete="name"
          />
          {fieldErrors.fullName && <small className={styles.fieldError}>{fieldErrors.fullName}</small>}
        </div>

        <div className="field">
          <label htmlFor="email">Email</label>
          <input
            id="email"
            type="email"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            autoComplete="email"
          />
          {fieldErrors.email && <small className={styles.fieldError}>{fieldErrors.email}</small>}
        </div>

        <div className="field">
          <label htmlFor="password">Пароль</label>
          <input
            id="password"
            type="password"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            autoComplete="new-password"
          />
          {fieldErrors.password && <small className={styles.fieldError}>{fieldErrors.password}</small>}
        </div>

        <button className="btn-primary" type="submit" disabled={submitting}>
          {submitting ? 'Создаём…' : 'Зарегистрироваться'}
        </button>

        <p className={styles.hint}>
          Уже есть аккаунт? <Link to="/login">Войдите</Link>
        </p>
      </form>
    </main>
  );
}
```

# Проверка

Запустите API и web:

```bash
# терминал 1
cd src/api/TaskPlanner.Api && dotnet run

# терминал 2
cd src/web && npm run dev
```

Проверьте руками:

1. Откройте `http://localhost:5173/tasks` → происходит переход на `/login`.
2. Нажмите «Зарегистрируйтесь», заполните форму, отправьте → попадаете на `/tasks`.
3. Обновите страницу (`F5`) → **остаётесь** на `/tasks`, а не на `/login`.
   Токен сохранился в `localStorage`.
4. Войдите с неверным паролем → видите «Неверный email или пароль».
5. Войдите с чужим email → то же сообщение, без подсказки, что email существует.
6. Зарегистрируйтесь с тем же email → «Пользователь с таким email уже существует».
7. Введите пароль из 5 символов → подсказка под полем «Пароль не короче 8 символов».
8. Проверьте в DevTools → Application → Local Storage:
   ключи `taskplanner.token` и `taskplanner.user`.

# проверить, что регистрация проходит через API, а не мимо него

# Если что-то не получилось

| Симптом | Что делать |
|---|---|
| После F5 снова `/login` | токен не сохранился в `localStorage` либо приложение стартует не с `/tasks` |
| `Cannot find module 'react-router-dom'` | `npm install react-router-dom` |
| «Failed to fetch» | API не запущен или прокси не настроен (страница 13) |
| Кириллица в поле даты съехала на день | используйте `toISOString()` — см. замечание на странице 17 |
| Ошибка в консоли про `Module '"react-router-dom"' has no exported member` | установлена v5, нужен `react-router-dom@7` |
| Регистрация прошла, но входа нет | в `RegisterPage` забыли вызвать `authApi.login` после регистрации |
