# 1. Хук загрузки списка (0–20 мин)

`src/web/src/hooks/useTasks.ts`:

```typescript
import { useCallback, useEffect, useRef, useState } from 'react';
import { errorMessage, tasksApi } from '../api/client';
import type { TaskQueryParams } from '../api/client';
import type { PagedResult, Task } from '../api/types';

export interface UseTasksResult {
  data: PagedResult<Task> | null;
  loading: boolean;
  error: string | null;
  reload: () => void;
}

export function useTasks(params: TaskQueryParams): UseTasksResult {
  const [data, setData] = useState<PagedResult<Task> | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  // Счётчик: пока идёт новый запрос, предыдущий результат помечается устаревшим
  const requestIdRef = useRef(0);

  // Строка параметров меняется на каждый рендер, если не склеивать вручную.
  // Склеиваем здесь — это ключ для useEffect.
  const key = JSON.stringify(params);

  const load = useCallback(() => {
    const requestId = ++requestIdRef.current;

    setLoading(true);
    setError(null);

    tasksApi
      .list(JSON.parse(key) as TaskQueryParams)
      .then((result) => {
        // Ответ устаревшего запроса игнорируем: пользователь уже сменил фильтр
        if (requestId !== requestIdRef.current) return;

        setData(result);
      })
      .catch((e: unknown) => {
        if (requestId !== requestIdRef.current) return;

        setData(null);
        setError(errorMessage(e));
      })
      .finally(() => {
        if (requestId !== requestIdRef.current) return;

        setLoading(false);
      });
  }, [key]);

  useEffect(() => {
    load();
  }, [load]);

  return { data, loading, error, reload: load };
}
```

> **Замечание:** без проверки `requestId` быстрые переключения фильтров дают
> «мигание»: ответ по старому фильтру приходит позже ответа по новому и
> перетирает его. Это самая частая ошибка в списках с фильтрами.

import type { Task, TaskPriority, TaskStatus } from '../api/types';
import styles from './TaskList.module.css';

const STATUS_LABELS: Record<TaskStatus, string> = {
  new: 'Новая',
  in_progress: 'В работе',
  done: 'Выполнена',
  cancelled: 'Отменена',
};

const PRIORITY_LABELS: Record<TaskPriority, string> = {
  high: 'Высокий',
  medium: 'Средний',
  low: 'Низкий',
};

export function formatDate(value: string | null): string {
  if (!value) return '—';

  // 2026-10-15 -> 15.10.2026
  const [year, month, day] = value.split('-');
  return `${day}.${month}.${year}`;
}

export function isOverdue(task: Task): boolean {
  if (task.status === 'done' || !task.dueDate) return false;

  const today = new Date().toISOString().slice(0, 10);
  return task.dueDate < today;
}

interface TaskListProps {
  items: Task[];
  onToggle: (task: Task) => void;
  onEdit: (task: Task) => void;
  onDelete: (task: Task) => void;
  busyId: number | null;
}

export function TaskList({ items, onToggle, onEdit, onDelete, busyId }: TaskListProps) {
  return (
    <ul className={styles.list}>
      {items.map((task) => (
        <li
          key={task.id}
          className={`${styles.item} ${isOverdue(task) ? styles.overdue : ''}`}
        >
          <label className={styles.checkbox}>
            <input
              type="checkbox"
              checked={task.status === 'done'}
              disabled={busyId === task.id}
              onChange={() => onToggle(task)}
            />
            <span className={styles.title}>{task.title}</span>
          </label>

          {task.description && <p className={styles.description}>{task.description}</p>}

          <div className={styles.meta}>
            <span className={`${styles.badge} ${styles[task.status]}`}>
              {STATUS_LABELS[task.status]}
            </span>
            <span className={`${styles.badge} ${styles[task.priority]}`}>
              {PRIORITY_LABELS[task.priority]}
            </span>
            <span className={styles.due}>
              Срок: {formatDate(task.dueDate)}
              {isOverdue(task) && <em className={styles.late}> — просрочена</em>}
            </span>
          </div>

          <div className={styles.actions}>
            <button type="button" className="btn-secondary" onClick={() => onEdit(task)}>
              Изменить
            </button>
            <button
              type="button"
              className="btn-secondary"
              onClick={() => onDelete(task)}
              disabled={busyId === task.id}
            >
              Удалить
            </button>
          </div>
        </li>
      ))}
    </ul>
  );
}

# 5. Пагинация (50–53 мин)

`src/web/src/components/Pagination.tsx`:

```tsx
interface PaginationProps {
  page: number;
  pageSize: number;
  total: number;
  pageSizeOptions: number[];
  onPageChange: (page: number) => void;
  onPageSizeChange: (pageSize: number) => void;
}

export function Pagination(props: PaginationProps) {
  const { page, pageSize, total, pageSizeOptions, onPageChange, onPageSizeChange } = props;

  const totalPages = pageSize > 0 ? Math.ceil(total / pageSize) : 0;
  const from = total === 0 ? 0 : (page - 1) * pageSize + 1;
  const to = Math.min(page * pageSize, total);

  return (
    <div className="pagination">
      <span className="counter">
        {total === 0 ? 'Нет записей' : `Показано ${from}–${to} из ${total}`}
      </span>

      <div className="controls">
        <label>
          На странице:
          <select
            value={pageSize}
            onChange={(e) => {
              onPageSizeChange(Number(e.target.value));
              onPageChange(1);
            }}
          >
            {pageSizeOptions.map((size) => (
              <option key={size} value={size}>{size}</option>
            ))}
          </select>
        </label>

        <button
          type="button"
          className="btn-secondary"
          disabled={page <= 1}
          onClick={() => onPageChange(page - 1)}
        >
          Назад
        </button>

        <span>Стр. {page} из {Math.max(totalPages, 1)}</span>

        <button
          type="button"
          className="btn-secondary"
          disabled={page >= totalPages}
          onClick={() => onPageChange(page + 1)}
        >
          Вперёд
        </button>
      </div>
    </div>
  );
}
```

# 6. Страница списка (53–60 мин)

`src/web/src/pages/TasksPage.tsx`:

```tsx
import { useCallback, useEffect, useState } from 'react';
import { Link, useNavigate } from 'react-router-dom';
import { errorMessage, tasksApi } from '../api/client';
import type { SortField, SortOrder, Task, TaskPriority, TaskStatus } from '../api/types';
import { useTasks } from '../hooks/useTasks';
import { Header } from '../components/Header';
import { Loader } from '../components/Loader';
import { EmptyState } from '../components/EmptyState';
import { Pagination } from '../components/Pagination';
import { TaskFilters, type FiltersValue } from '../components/TaskFilters';
import { TaskList } from '../components/TaskList';
import styles from './TasksPage.module.css';

const INITIAL_FILTERS: FiltersValue = {
  status: '',
  priority: '',
  q: '',
  sort: 'createdAt',
  order: 'desc',
};

const PAGE_SIZES = [5, 10, 20, 50];

export function TasksPage() {
  const navigate = useNavigate();

  // Фильтры в форме (ещё не применены) и применённые — разные состояния,
  // иначе список перезапрашивается на каждое нажатие клавиши
  const [draft, setDraft] = useState<FiltersValue>(INITIAL_FILTERS);
  const [applied, setApplied] = useState<FiltersValue>(INITIAL_FILTERS);
  const [page, setPage] = useState(1);
  const [pageSize, setPageSize] = useState(20);
  const [busyId, setBusyId] = useState<number | null>(null);
  const [actionError, setActionError] = useState<string | null>(null);

  const { data, loading, error, reload } = useTasks({
    status: (applied.status || undefined) as TaskStatus | undefined,
    priority: (applied.priority || undefined) as TaskPriority | undefined,
    q: applied.q || undefined,
    sort: applied.sort,
    order: applied.order,
    page,
    pageSize,
  });

  // Поиск с задержкой 400 мс — отдельный эффект, независимый от кнопки
  useEffect(() => {
    const timer = setTimeout(() => {
      setApplied((prev) => (prev.q === draft.q ? prev : { ...prev, q: draft.q }));
      setPage(1);
    }, 400);

    return () => clearTimeout(timer);
  }, [draft.q]);

  const hasActiveFilters =
    applied.status !== '' ||
    applied.priority !== '' ||
    applied.q !== '' ||
    applied.sort !== 'createdAt' ||
    applied.order !== 'desc';

  function handleApply() {
    setApplied(draft);
    setPage(1);
  }

  function handleReset() {
    setDraft(INITIAL_FILTERS);
    setApplied(INITIAL_FILTERS);
    setPage(1);
  }

  const handleToggle = useCallback(async (task: Task) => {
    setBusyId(task.id);
    setActionError(null);

    try {
      await tasksApi.toggleCompleted(task.id);
      reload();
    } catch (e) {
      setActionError(errorMessage(e));
    } finally {
      setBusyId(null);
    }
  }, [reload]);

  const handleDelete = useCallback(async (task: Task) => {
    if (!window.confirm(`Удалить задачу «${task.title}»?`)) return;

    setBusyId(task.id);
    setActionError(null);

    try {
      await tasksApi.remove(task.id);
      reload();
    } catch (e) {
      setActionError(errorMessage(e));
    } finally {
      setBusyId(null);
    }
  }, [reload]);

  return (
    <div>
      <Header />

      <main className={styles.content}>
        <div className={styles.head}>
          <h1>Задачи</h1>
          <button type="button" className="btn-primary" onClick={() => navigate('/tasks/new')}>
            Новая задача
          </button>
        </div>

        <TaskFilters
          value={draft}
          onChange={setDraft}
          onApply={handleApply}
          onReset={handleReset}
          onSearchText={(q) => setDraft((prev) => ({ ...prev, q }))}
          hasActiveFilters={hasActiveFilters}
        />

        {actionError && <div className="error" role="alert">{actionError}</div>}
        {error && <div className="error" role="alert">{error}</div>}

        {loading && <Loader />}

        {!loading && data && data.items.length === 0 && (
          <EmptyState
            title={hasActiveFilters ? 'Ничего не найдено' : 'Задач пока нет'}
            hint={hasActiveFilters ? 'Измените фильтры или сбросьте их' : 'Создайте первую задачу'}
            action={
              hasActiveFilters ? (
                <button type="button" className="btn-secondary" onClick={handleReset}>
                  Сбросить фильтры
                </button>
              ) : (
                <button type="button" className="btn-primary" onClick={() => navigate('/tasks/new')}>
                  Новая задача
                </button>
              )
            }
          />
        )}

        {!loading && data && data.items.length > 0 && (
          <TaskList
            items={data.items}
            onToggle={handleToggle}
            onEdit={(task) => navigate(`/tasks/${task.id}/edit`)}
            onDelete={handleDelete}
            busyId={busyId}
          />
        )}

        {!loading && data && (
          <Pagination
            page={data.page}
            pageSize={data.pageSize}
            total={data.total}
            pageSizeOptions={PAGE_SIZES}
            onPageChange={setPage}
            onPageSizeChange={setPageSize}
          />
        )}
      </main>
    </div>
  );
}
```

# Проверка

Подготовьте данные: `db/seed.sql` плюс 40 задач из страницы 11.

1. Откройте `/tasks` → список из 10–50 задач, пагинация показывает «Показано 1–20 из 50».
2. Смените «На странице» на 5 → «Показано 1–5 из 50», «Стр. 1 из 10».
3. Нажмите «Вперёд» три раза → первая задача четвёртой страницы; первой страницы
   задачи не повторяются.
4. Выберите статус «Выполненные» + «Применить» → в списке только `done`.
5. Введите в поиск «ER» → список сокращается. Введите «щщщ» → появляется
   «Ничего не найдено» и кнопка «Сбросить фильтры».
6. Поставьте «На странице: 50» и примените статус «Новые» → все карточки `new`.
7. Поставьте сортировку «По сроку», порядок «По возрастанию» → задачи без срока
   уходят в конец, просроченные отмечены красной полосой и подписью «просрочена».
8. Отметьте задачу галочкой → она становится «Выполнена», счётчик на пагинации
   уменьшается после перезагрузки списка.
9. Нажмите «Удалить» → диалог подтверждения; отмена ничего не удаляет,
   подтверждение убирает задачу.
10. Сузьте окно до 360 px → фильтры складываются в одну колонку, кнопки
    переносятся, горизонтальной прокрутки нет.

# Если что-то не получилось

| Симптом | Что делать |
|---|---|
| Список «прыгает» при переключении фильтров | не проверяется `requestId` в `useTasks` |
| Поиск срабатывает на каждую букву и тормозит | нет `setTimeout`/debounce либо таймер не чистится в `useEffect` |
| `useTasks` бесконечно перезапрашивает | объект `params` меняется каждый рендер; нужен `key = JSON.stringify(params)` |
| «Показано 1–20 из 0» | `totalPages` считается как `total / pageSize` с округлением вниз; используйте `Math.ceil` |
| Кнопка «Вперёд» активна на последней странице | не проверяется `page >= totalPages` |
| `DELETE` возвращает HTML, а не пустоту | в `tasksApi.remove` используйте `await api.delete(...)` без обращения к `response.data` |
| Просроченные не подсвечены | `isOverdue` сравнивает строки дат; `2026-10-01 < 2026-10-02` в JS работает, но пустая дата даёт `true` — проверьте `!task.dueDate` |
