using TaskPlanner.Core.Enums;

namespace TaskPlanner.Core.Dtos;

/// <summary>
/// Параметры GET /api/tasks. Значения по умолчанию совпадают с контрактом
/// api/openapi.yaml: sort=createdAt, order=desc, page=1, pageSize=20.

public class TaskQuery
{
    public TaskStatus? Status { get; set; }

    public TaskPriority? Priority { get; set; }

    /// <summary>Фильтр по точному сроку: задачи на эту дату.</summary>

    /// <summary>asc | desc</summary>
    public string Order { get; set; } = "desc";

    public int Page { get; set; } = 1;

    public int PageSize { get; set; } = 20;
}

# 2. Разбор и валидация параметров (95–110 мин)

Строку `?sort=hash` нельзя молча проигнорировать: клиент думает, что сортирует
по одному, а получает по другому. Неизвестное значение — `400`.

`src/api/TaskPlanner.Api/Services/TaskQueryParser.cs`:

```csharp
using System.Text.RegularExpressions;
using TaskPlanner.Core.Dtos;
using TaskPlanner.Core.Enums;
using TaskPlanner.Core.Exceptions;

namespace TaskPlanner.Api.Services;

public static partial class TaskQueryParser
{
    private static readonly string[] AllowedSort =
        ["createdAt", "dueDate", "priority", "title"];

    public static TaskQuery Parse(
        string? status,
        string? priority,
        string? dueDate,
        string? q,
        string? sort,
        string? order,
        int? page,
        int? pageSize)
    {
        var query = new TaskQuery
        {
            Status = ParseStatus(status),
            Priority = ParsePriority(priority),
            DueDate = ParseDate(dueDate),
            Q = string.IsNullOrWhiteSpace(q) ? null : q.Trim(),
            Page = page ?? 1,
            PageSize = pageSize ?? 20,
        };

        query.Sort = string.IsNullOrWhiteSpace(sort) ? "createdAt" : sort.Trim();
        if (!AllowedSort.Contains(query.Sort, StringComparer.Ordinal))
        {
            throw ApiException.BadRequest(
                $"Неизвестный параметр сортировки: {query.Sort}. " +
                $"Допустимо: {string.Join(", ", AllowedSort)}");
        }

        query.Order = string.IsNullOrWhiteSpace(order) ? "desc" : order.Trim().ToLowerInvariant();
        if (query.Order is not ("asc" or "desc"))
        {
            throw ApiException.BadRequest("Неизвестный порядок сортировки: допустимо asc или desc");
        }

        if (query.Page < 1)
        {
            throw ApiException.BadRequest("Номер страницы не может быть меньше 1");
        }

        if (query.PageSize is < 1 or > 100)
        {
            throw ApiException.BadRequest("Размер страницы должен быть от 1 до 100");
        }

        return query;
    }

    private static TaskStatus? ParseStatus(string? value)
    {
        if (string.IsNullOrWhiteSpace(value))
        {
            return null;
        }

        return value.Trim() switch
        {
            "new" => TaskStatus.New,
            "in_progress" => TaskStatus.InProgress,
            "done" => TaskStatus.Done,
            "cancelled" => TaskStatus.Cancelled,
            _ => throw ApiException.BadRequest(
                $"Неизвестный статус: {value}. Допустимо: new, in_progress, done, cancelled"),
        };
    }

    private static TaskPriority? ParsePriority(string? value)
    {
        if (string.IsNullOrWhiteSpace(value))
        {
            return null;
        }

        return value.Trim() switch
        {
            "low" => TaskPriority.Low,
            "medium" => TaskPriority.Medium,
            "high" => TaskPriority.High,
            _ => throw ApiException.BadRequest(
                $"Неизвестный приоритет: {value}. Допустимо: low, medium, high"),
        };
    }

    private static DateOnly? ParseDate(string? value)
    {
        if (string.IsNullOrWhiteSpace(value))
        {
            return null;
        }

        // Строгий разбор: 'yyyy-MM-dd'. DateOnly.TryParseExact не принимает
        // '01.10.2026' — это осознанно, чтобы клиенты не путали форматы.
        if (DateOnly.TryParseExact(
                value.Trim(),
                "yyyy-MM-dd",
                System.Globalization.CultureInfo.InvariantCulture,
                System.Globalization.DateTimeStyles.None,
                out var date))
        {
            return date;
        }

        throw ApiException.BadRequest(
            $"Дата должна быть в формате ГГГГ-ММ-ДД, получено: {value}");
    }

    /// <summary>
    /// Экранирует спецсимволы LIKE, чтобы пользовательский ввод
    /// «100%» не превратился в шаблон поиска.
    /// </summary>
    public static string EscapeLike(string input)
        => WildcardRegex().Replace(input, m => @"\" + m.Value);

    [GeneratedRegex(@"[%_\\]")]
    private static partial Regex WildcardRegex();
}
```

> **Замечание:** `EscapeLike` обязателен. Без него запрос `?q=%` превратится в
> «найти всё», а `?q=_` — в «найти задачи из одного символа». Экранирование
> требует ещё и `ESCAPE '\'` в выражении `ILIKE` — ниже он есть.

# 3. Запрос со всеми фильтрами (110–125 мин)

`src/api/TaskPlanner.Api/Services/TaskService.Search.cs`:

```csharp
using Microsoft.EntityFrameworkCore;
using TaskPlanner.Api.Mappers;
using TaskPlanner.Core.Common;
using TaskPlanner.Core.Dtos;
using TaskPlanner.Core.Entities;

namespace TaskPlanner.Api.Services;

public partial class TaskService
{
    public async Task<PagedResult<TaskDto>> SearchAsync(
        long userId,
        TaskQuery query,
        CancellationToken cancellationToken)
    {
        // Один предикат переиспользуется дважды: для total и для выборки
        // страницы. Копировать условия в два места нельзя — они разойдутся.
        IQueryable<TaskItem> BuildQuery()
        {
            var queryable = db.Tasks.AsNoTracking().Where(x => x.UserId == userId);

            if (query.Status is not null)
            {
                queryable = queryable.Where(x => x.Status == query.Status.Value);
            }

            if (query.Priority is not null)
            {
                queryable = queryable.Where(x => x.Priority == query.Priority.Value);
            }

            if (query.DueDate is not null)
            {
                queryable = queryable.Where(x => x.DueDate == query.DueDate.Value);
            }

            if (query.Q is not null)
            {
                var pattern = $"%{TaskQueryParser.EscapeLike(query.Q)}%";

                // ILIKE — регистронезависимый LIKE. ESCAPE явно указан,
                // чтобы символ «\» из EscapeLike работал как escape-символ.
                queryable = queryable.Where(x =>
                    EF.Functions.ILike(x.Title, pattern, @"\")
                    || (x.Description != null
                        && EF.Functions.ILike(x.Description, pattern, @"\")));
            }

            return queryable;
        }

        // total считается отдельным запросом по тому же предикату.
        // Без Skip/Take: мы считаем ВСЕ подходящие строки, а не загружаем их.
        var total = await BuildQuery().LongCountAsync(cancellationToken);

        var ascending = query.Order == "asc";

        var sorted = query.Sort switch
        {
            "dueDate" => SortByDueDate(BuildQuery(), ascending),
            "priority" => ascending
                ? BuildQuery().OrderBy(x => x.Priority).ThenBy(x => x.Id)
                : BuildQuery().OrderByDescending(x => x.Priority).ThenBy(x => x.Id),
            "title" => ascending
                ? BuildQuery().OrderBy(x => x.Title).ThenBy(x => x.Id)
                : BuildQuery().OrderByDescending(x => x.Title).ThenBy(x => x.Id),
            _ => ascending
                ? BuildQuery().OrderBy(x => x.CreatedAt).ThenBy(x => x.Id)
                : BuildQuery().OrderByDescending(x => x.CreatedAt).ThenBy(x => x.Id),
        };

        var items = await sorted
            .Skip((query.Page - 1) * query.PageSize)
            .Take(query.PageSize)
            .ToListAsync(cancellationToken);

        return new PagedResult<TaskDto>
        {
            Items = items.Select(x => x.ToDto()).ToList(),
            Total = total,
            Page = query.Page,
            PageSize = query.PageSize,
        };
    }

    /// <summary>
    /// Задачи без срока уходят в конец при любом порядке: пустой срок —
    /// это «неважно», а не «важно прямо сейчас». NULLS LAST в PostgreSQL
    /// выражается двумя OrderBy, потому что провайдер Npgsql не умеет
    /// NULLS FIRST/LAST через LINQ.
    /// </summary>
    private static IOrderedQueryable<TaskItem> SortByDueDate(
        IQueryable<TaskItem> queryable,
        bool ascending)
    {
        return ascending
            ? queryable.OrderBy(x => x.DueDate == null).ThenBy(x => x.DueDate).ThenBy(x => x.Id)
            : queryable.OrderByDescending(x => x.DueDate == null).ThenBy(x => x.DueDate).ThenBy(x => x.Id);
    }
}
```

> **Замечание:** `TaskQueryParser` и `TaskService` лежат в одном пространстве
> имён `TaskPlanner.Api.Services`, поэтому дополнительных `using` не требуется.
> `PagedResult<T>` лежит в `TaskPlanner.Core.Common` — если компилятор его не
> находит, добавьте `using TaskPlanner.Core.Common;` в файл.

Добавьте ещё:

```sql
INSERT INTO tasks (user_id, title, status, priority, due_date, completed_at)
SELECT
    1,
    'Задача №' || g,
    CASE WHEN g % 3 = 0 THEN 'done' ELSE 'new' END,
    CASE g % 3 WHEN 0 THEN 'low' WHEN 1 THEN 'medium' ELSE 'high' END,
    CURRENT_DATE + (g % 14),
    CASE WHEN g % 3 = 0 THEN NOW() ELSE NULL END
FROM generate_series(1, 40) AS g;
```

Обратите внимание на `generate_series` — способ сгенерировать 40 задач без
40 строк `INSERT`. `CASE WHEN` в том же стиле, что и `FILTER` в сводке.

# Если что-то не получилось

| Симптом | Что делать |
|---|---|
| `Skip((Page-1)*PageSize)` даёт отрицательный индекс | не проверили `Page < 1` в парсере |
| `?q=%` находит всё | не вызвали `EscapeLike` либо забыли `ESCAPE '\'` |
| `dueDate=01.10.2026` не даёт `400` | используете `DateTime.TryParse` вместо `TryParseExact` |
| Сортировка по сроку начинается с `null` | не сработала пара `OrderBy(x => x.DueDate == null).ThenBy(...)` |
| `total` не совпадает с числом строк | считаете `total` после `Take`, а нужно `LongCountAsync` без пагинации |
| ILIKE не экранирует `_` | забыли `EscapeLike` для `q` либо не указали escape-символ в выражении |
| Страница 2 совпадает со страницей 1 | не вызвали `Skip` |
| `status=Done` (с большой буквы) даёт `400` | контракт требует строчные значения: `new`, `in_progress`, `done`, `cancelled` |
