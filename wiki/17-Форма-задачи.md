Сессия 6. Web-клиент: форма задачи · 60–110 мин

# 1. Маршруты формы (60–63 мин)

В `App.tsx` добавьте два маршрута внутрь `RequireAuth`:

```tsx
<Route path="/tasks/new" element={<RequireAuth><TaskFormPage /></RequireAuth>} />
<Route path="/tasks/:id/edit" element={<RequireAuth><TaskFormPage /></RequireAuth>} />
```

Добавьте импорт:

```tsx
import { TaskFormPage } from './pages/TaskFormPage';
```

Порядок важен: `/tasks/new` должен быть объявлен **до**
`/tasks/:id/edit`, иначе `new` будет разобран как `id = "new"`.

> **Замечание:** в react-router 7 маршруты ранжируются по специфичности, и
> статический сегмент `new` приоритетнее динамического `:id`. Явный порядок
> в `Routes` подстраховывает и делает намерение читаемым.

# 2. Форма (63–95 мин)

`src/web/src/pages/TaskFormPage.tsx`:

```tsx
import { useEffect, useState, type FormEvent } from 'react';
import { useNavigate, useParams } from 'react-router-dom';
import { errorMessage, tasksApi } from '../api/client';
import type { Task, TaskPriority, TaskStatus } from '../api/types';
import { Header } from '../components/Header';
import { Loader } from '../components/Loader';
import styles from './TaskFormPage.module.css';

interface FormValue {
  title: string;
  description: string;
  priority: TaskPriority;
  status: TaskStatus;
  dueDate: string;
  categoryId: string;
}

const EMPTY: FormValue = {
  title: '',
  description: '',
  priority: 'medium',
  status: 'new',
  dueDate: '',
  categoryId: '',
};

/** 2026-10-15 -> локальная дата для <input type="date">. */
function isoToInputDate(value: string | null): string {
  if (!value) return '';
  return value.slice(0, 10);
}

export function TaskFormPage() {
  const navigate = useNavigate();
  const params = useParams<{ id: string }>();

  const id = params.id ? Number(params.id) : null;
  const isEdit = id !== null && Number.isFinite(id);

  const [value, setValue] = useState<FormValue>(EMPTY);
  const [loading, setLoading] = useState(isEdit);
  const [submitting, setSubmitting] = useState(false);
  const [deleting, setDeleting] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [titleError, setTitleError] = useState<string | null>(null);

  useEffect(() => {
    if (!isEdit) {
      setValue(EMPTY);
      return;
    }

    let cancelled = false;

    tasksApi
      .getById(id)
      .then((task: Task) => {
        if (cancelled) return;

        setValue({
          title: task.title,
          description: task.description ?? '',
          priority: task.priority,
          status: task.status,
          dueDate: isoToInputDate(task.dueDate),
          categoryId: task.categoryId === null ? '' : String(task.categoryId),
        });
      })
      .catch((e: unknown) => {
        if (cancelled) return;
        setError(errorMessage(e));
      })
      .finally(() => {
        if (!cancelled) setLoading(false);
      });

    return () => {
      cancelled = true;
    };
  }, [id, isEdit]);

  function validate(): boolean {
    const title = value.title.trim();

    if (!title) {
      setTitleError('Название обязательно');
      return false;
    }

    if (title.length > 200) {
      setTitleError('Название не длиннее 200 символов');
      return false;
    }

    setTitleError(null);
    return true;
  }

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setError(null);

    if (!validate()) return;

    const payload = {
      title: value.title.trim(),
      description: value.description.trim() === '' ? null : value.description.trim(),
      priority: value.priority,
      status: value.status,
      dueDate: value.dueDate === '' ? null : value.dueDate,
      categoryId: value.categoryId === '' ? null : Number(value.categoryId),
    };

    setSubmitting(true);

    try {
      if (isEdit) {
        await tasksApi.update(id, payload);
      } else {
        await tasksApi.create(payload);
      }

      navigate('/tasks');
    } catch (e) {
      setError(errorMessage(e));
    } finally {
      setSubmitting(false);
    }
  }

  async function handleDelete() {
    if (!isEdit) return;
    if (!window.confirm(`Удалить задачу «${value.title}»?`)) return;

    setDeleting(true);
    setError(null);

    try {
      await tasksApi.remove(id);
      navigate('/tasks');
    } catch (e) {
      setError(errorMessage(e));
      setDeleting(false);
    }
  }

  return (
    <div>
      <Header />

      <main className={styles.content}>
        <h1>{isEdit ? 'Изменить задачу' : 'Новая задача'}</h1>

        {loading && <Loader />}

        {!loading && (
          <form className={styles.form} onSubmit={handleSubmit} noValidate>
            {error && <div className="error" role="alert">{error}</div>}

            <div className="field">
              <label htmlFor="title">Название *</label>
              <input
                id="title"
                value={value.title}
                maxLength={200}
                onChange={(e) => setValue((prev) => ({ ...prev, title: e.target.value }))}
                placeholder="Что нужно сделать"
              />
              {titleError && <small className={styles.fieldError}>{titleError}</small>}
            </div>

            <div className="field">
              <label htmlFor="description">Описание</label>
              <textarea
                id="description"
                rows={4}
                value={value.description}
                onChange={(e) => setValue((prev) => ({ ...prev, description: e.target.value }))}
              />
            </div>

            <div className={styles.row}>
              <div className="field">
                <label htmlFor="priority">Приоритет</label>
                <select
                  id="priority"
                  value={value.priority}
                  onChange={(e) =>
                    setValue((prev) => ({ ...prev, priority: e.target.value as TaskPriority }))
                  }
                >
                  <option value="low">Низкий</option>
                  <option value="medium">Средний</option>
                  <option value="high">Высокий</option>
                </select>
              </div>

              <div className="field">
                <label htmlFor="status">Статус</label>
                <select
                  id="status"
                  value={value.status}
                  onChange={(e) =>
                    setValue((prev) => ({ ...prev, status: e.target.value as TaskStatus }))
                  }
                >
                  <option value="new">Новая</option>
                  <option value="in_progress">В работе</option>
                  <option value="done">Выполнена</option>
                  <option value="cancelled">Отменена</option>
                </select>
              </div>
            </div>

            <div className={styles.row}>
              <div className="field">
                <label htmlFor="dueDate">Срок</label>
                <input
                  id="dueDate"
                  type="date"
                  value={value.dueDate}
                  onChange={(e) => setValue((prev) => ({ ...prev, dueDate: e.target.value }))}
                />
              </div>

              <div className="field">
                <label htmlFor="categoryId">Категория (id)</label>
                <input
                  id="categoryId"
                  type="number"
                  min={1}
                  value={value.categoryId}
                  onChange={(e) => setValue((prev) => ({ ...prev, categoryId: e.target.value }))}
                  placeholder="необязательно"
                />
              </div>
            </div>

            <p className={styles.note}>
              Категории в задании необязательны и за них не начисляют баллы —
              поле можно оставить пустым.
            </p>

            <div className={styles.actions}>
              <button className="btn-primary" type="submit" disabled={submitting || deleting}>
                {submitting ? 'Сохраняем…' : 'Сохранить'}
              </button>

              <button
                className="btn-secondary"
                type="button"
                onClick={() => navigate('/tasks')}
                disabled={submitting || deleting}
              >
                Отмена
              </button>

              {isEdit && (
                <button
                  className={styles.danger}
                  type="button"
                  onClick={handleDelete}
                  disabled={submitting || deleting}
                >
                  {deleting ? 'Удаляем…' : 'Удалить'}
                </button>
              )}
            </div>
          </form>
        )}
      </main>
    </div>
  );
}
```

# 3. Ловушка с датой (95–100 мин)

`Date` в JavaScript работает в **UTC**, а `<input type="date">` — в
**локальном** времени. Из-за этого самая частая ошибка — задача сдвигается на
один день.

Неправильно:

```typescript
// Если сейчас 2026-10-15 в 22:00 по Москве, то в UTC уже 2026-10-15T19:00,
// а если в 03:00 по Москве — то UTC отстаёт, и дата уедет назад на сутки
new Date().toISOString().slice(0, 10); // неверно для input[type=date]
```

Правильно — брать локальные части даты:

```typescript
function toInputDate(date: Date): string {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');

  return `${year}-${month}-${day}`;
}
```

Для обратного преобразования `toISOString()` **безопасен** — сервер отдаёт
`"2026-10-15T00:00:00+00:00"`, и `.slice(0, 10)` даёт `2026-10-15`.

Проверьте ловушку: создайте задачу со сроком «сегодня», откройте её на
редактирование — дата должна совпадать, а не отличаться на сутки.

# 4. Статус и дата выполнения (100–103 мин)

В форме есть выбор статуса, и он напрямую влияет на `completed_at`:

| Что прислал клиент | Что делает сервер |
|---|---|
| `status = "done"` | выставляет `completedAt = now` |
| любой другой статус | `completedAt = null` |

Если бы клиент позволял вручную выставить «дату выполнения», она разошлась бы
со статусом и `CHECK ck_tasks_completed_at` отклонил бы сохранение. Правило
«дата выполнения вычисляется сервером» снимает эту проблему целиком.

Отдельно: галочка «выполнено» в списке (страница 16) вызывает
`PATCH /tasks/{id}/complete`, а форма — `PUT /tasks/{id}`. Оба пути приводят к
одному состоянию, проверка на сервере общая.

# 5. Обновите `App.tsx` целиком (103–110 мин)

```tsx
import { BrowserRouter, Route, Routes } from 'react-router-dom';
import { AuthProvider } from './auth/AuthContext';
import { RequireAuth, RequireGuest } from './auth/RequireAuth';
import { LoginPage } from './pages/LoginPage';
import { RegisterPage } from './pages/RegisterPage';
import { TasksPage } from './pages/TasksPage';
import { TaskFormPage } from './pages/TaskFormPage';
import { SummaryPage } from './pages/SummaryPage';

export function App() {
  return (
    <BrowserRouter>
      <AuthProvider>
        <Routes>
          <Route path="/login" element={<RequireGuest><LoginPage /></RequireGuest>} />
          <Route path="/register" element={<RequireGuest><RegisterPage /></RequireGuest>} />

          {/* Статический сегмент new объявлен до динамического :id */}
          <Route path="/tasks/new" element={<RequireAuth><TaskFormPage /></RequireAuth>} />
          <Route path="/tasks/:id/edit" element={<RequireAuth><TaskFormPage /></RequireAuth>} />

          <Route path="/tasks" element={<RequireAuth><TasksPage /></RequireAuth>} />
          <Route path="/summary" element={<RequireAuth><SummaryPage /></RequireAuth>} />

          <Route path="*" element={<RequireAuth><TasksPage /></RequireAuth>} />
        </Routes>
      </AuthProvider>
    </BrowserRouter>
  );
}
```

# Проверка

1. Нажмите «Новая задача» → форма пустая, заголовок «Новая задача», кнопки
   «Удалить» нет.
2. Отправьте пустую форму → «Название обязательно» под полем, запрос не ушёл
   (проверьте Network — запросов нет).
3. Введите 201 символ → ограничение `maxLength` не даёт ввести больше 200.
4. Заполните всё, сохраните → переход на `/tasks`, задача есть в списке.
5. Откройте задачу на редактирование → все поля заполнены из API, дата в
   том же дне, что была сохранена.
6. Смените статус на «Выполнена», сохраните, откройте заново → статус
   `done`, в базе `completed_at` заполнен:

```sql
SELECT id, title, status, completed_at FROM tasks
WHERE title = 'Что-то новое' ORDER BY id DESC LIMIT 1;
```

7. Смените статус обратно на «Новая» → `completed_at` снова `NULL`:

```sql
SELECT status, completed_at IS NULL AS completed_at_is_null
FROM tasks ORDER BY id DESC LIMIT 1;
-- new | t
```

8. Поставьте срок «сегодня» → сохраните, откройте заново → дата не сдвинулась.
9. Нажмите «Удалить» → отмена ничего не удаляет; подтверждение убирает задачу
   и возвращает на список.
10. Откройте `/tasks/999999/edit` чужой или несуществующей задачи →
    сообщение «Задача не найдена», форма не заполняется.

```bash
# клиентский код не должен подставлять categoryId, которого нет у пользователя
curl -s -X POST http://localhost:5173/api/tasks \
  -H "Authorization: Bearer $TOKEN" -H 'Content-Type: application/json' \
  -d '{"title":"С чужой категорией","categoryId":9999}' \
  -i | head -1
# ожидается 500 или 400 — нарушение внешнего ключа.
# Это повод сделать categoryId всегда null, если список категорий не реализован.
```

# Если что-то не получилось

| Симптом | Что делать |
|---|---|
| `/tasks/new` открывает форму редактирования | `new` объявлен после `/tasks/:id/edit`, либо `Number('new')` = `NaN` и `isEdit` вычисляется верно — проверьте порядок маршрутов |
| При открытии существующей задачи поля пустые | `useEffect` не отработал: неверна зависимость `[id, isEdit]` или отменён `cancelled` |
| Дата уехала на сутки | используется `toISOString()` для `input[type=date]` |
| `400 invalid_error` при сохранении | `title` длиннее 200 или пустой; смотрите текст `message` в ответе |
| `500` при сохранении с `categoryId` | такого `categoryId` нет; отправляйте `null` |
| После сохранения остались на форме | нет `navigate('/tasks')` в блоке `try` после успеха |
| Кнопка «Удалить» видна при создании | условие `{isEdit && …}` проверяет `isEdit`, а не `id !== null` |
