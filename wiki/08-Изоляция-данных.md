# 1. Кто я (95–105 мин)

Токен уже содержит идентификатор пользователя в claim `sub`. Достаём его
из текущего запроса:

`src/api/TaskPlanner.Api/Infrastructure/ICurrentUser.cs`:

```csharp
namespace TaskPlanner.Api.Infrastructure;

/// <summary>
/// Текущий пользователь, определённый по JWT.
/// Позволяет сервисам не зависеть от HttpContext.
/// </summary>
public interface ICurrentUser
{
    /// <summary>Идентификатор пользователя. Бросает исключение, если токена нет.</summary>
    long GetUserId();

    /// <summary>Email из токена. Может быть пустым.</summary>
    string? GetEmail();
}
```

`src/api/TaskPlanner.Api/Infrastructure/CurrentUserAccessor.cs`:

```csharp
using System.Security.Claims;
using TaskPlanner.Core.Exceptions;

namespace TaskPlanner.Api.Infrastructure;

public class CurrentUserAccessor(IHttpContextAccessor httpContextAccessor) : ICurrentUser
{
    public long GetUserId()
    {
        var principal = httpContextAccessor.HttpContext?.User;

        var value = principal?.FindFirst("sub")?.Value
                    ?? principal?.FindFirst(ClaimTypes.NameIdentifier)?.Value;

        if (!long.TryParse(value, out var userId))
        {
            throw ApiException.Unauthorized("Требуется авторизация");
        }

        return userId;
    }

    public string? GetEmail()
        => httpContextAccessor.HttpContext?.User?.FindFirst("email")?.Value;
}
```

В `Program.cs` добавьте регистрацию до `builder.Build()`:

```csharp
builder.Services.AddHttpContextAccessor();
builder.Services.AddScoped<ICurrentUser, CurrentUserAccessor>();
```

# 2. Правильный код для чужого объекта (105–115 мин)

Это главная мысль сессии. Сравните два варианта:

| Что делает пользователь | Ответ `403 Forbidden` | Ответ `404 Not Found` |
|---|---|---|
| Просит задачу, которая его | 200 | 200 |
| Просит задачу, которая **чужая** | «Доступ запрещён» | «Задача не найдена» |
| Просит id `999999`, которого нет | «Доступ запрещён» | «Задача не найдена» |

С `403` по ответам можно **перечислить чужие объекты**: «задача 7 существует,
но вам нельзя» против «задачи 7 не существует». Значит id можно подбирать и
узнавать содержимое базы другого человека. С `404` ответы неразличимы.

Правило: **чужие объекты не выдаём как существующие**. `403` оставляем для
ситуации «роль не позволяет действие» — например, администраторский метод,
который обычному пользователю недоступен.

`src/api/TaskPlanner.Api/Services/TaskService.cs`:

```csharp
using Microsoft.EntityFrameworkCore;
using TaskPlanner.Api.Data;
using TaskPlanner.Api.Mappers;
using TaskPlanner.Core.Dtos;
using TaskPlanner.Core.Entities;
using TaskPlanner.Core.Exceptions;

namespace TaskPlanner.Api.Services;

public partial class TaskService(AppDbContext db, ILogger<TaskService> logger) : ITaskService
{
    public async Task<IReadOnlyList<TaskDto>> GetAllAsync(
        long userId,
        CancellationToken cancellationToken)
    {
        // user_id в предикате — единственное, что делает выборку безопасной.
        // Забыть его здесь = утечка чужих данных.
        var tasks = await db.Tasks
            .AsNoTracking()
            .Where(x => x.UserId == userId)
            .OrderByDescending(x => x.CreatedAt)
            .ToListAsync(cancellationToken);

        return tasks.Select(x => x.ToDto()).ToList();
    }

    public async Task<TaskDto> GetByIdAsync(
        long userId,
        long id,
        CancellationToken cancellationToken)
    {
        // Один запрос с двумя условиями: и id, и владелец.
        var task = await db.Tasks
            .AsNoTracking()
            .FirstOrDefaultAsync(x => x.Id == id && x.UserId == userId, cancellationToken);

        if (task is null)
        {
            // Не различаем «нет такой задачи» и «это не ваша задача».
            throw ApiException.NotFound("Задача не найдена");
        }

        return task.ToDto();
    }

    public async Task<TaskDto> CreateAsync(
        long userId,
        TaskInput input,
        CancellationToken cancellationToken)
    {
        var now = DateTimeOffset.UtcNow;

        var task = new TaskItem
        {
            UserId = userId,
            Title = input.Title.Trim(),
            Description = string.IsNullOrWhiteSpace(input.Description)
                ? null
                : input.Description.Trim(),
            Status = input.Status,
            Priority = input.Priority,
            DueDate = input.DueDate,
            CategoryId = input.CategoryId,

            // CHECK ck_tasks_completed_at требует согласованности:
            // completed_at заполняется ТОЛЬКО при статусе done.
            CompletedAt = input.Status == Enums.TaskStatus.Done ? now : null,

            CreatedAt = now,
            UpdatedAt = now,
        };

        db.Tasks.Add(task);
        await db.SaveChangesAsync(cancellationToken);

        logger.LogInformation(
            "Пользователь {UserId} создал задачу {TaskId}", userId, task.Id);

        return task.ToDto();
    }
}
```

> **Замечание:** `AsNoTracking()` обязателен для чтения. Без него EF отслеживает
> каждую прочитанную сущность в `ChangeTracker` — растёт память, а при
> `SaveChangesAsync` случайно можно перезаписать чужой объект. Для методов,
> которые меняют данные, `AsNoTracking` наоборот **не** нужен.

> **Замечание:** `CompletedAt` выставляется **до** сохранения, в том же объекте.
> Если сделать два `SaveChangesAsync` — по одному на статус и на дату —
> промежуточное состояние нарушит `ck_tasks_completed_at`, и база вернёт ошибку.

# 4. Защищённый контроллер (140–150 мин)

`src/api/TaskPlanner.Api/Controllers/TasksController.cs`:

```csharp
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Swashbuckle.AspNetCore.Annotations;
using TaskPlanner.Api.Infrastructure;
using TaskPlanner.Api.Services;
using TaskPlanner.Core.Common;
using TaskPlanner.Core.Dtos;

namespace TaskPlanner.Api.Controllers;

[ApiController]
[Route("api/tasks")]
[Authorize]
public class TasksController(ITaskService taskService, ICurrentUser currentUser) : ControllerBase
{
    /// <summary>Список задач текущего пользователя.</summary>
    [HttpGet]
    [ProducesResponseType(typeof(ApiResponse<IReadOnlyList<TaskDto>>), StatusCodes.Status200OK)]
    [ProducesResponseType(typeof(ApiResponse<object?>), StatusCodes.Status401Unauthorized)]
    public async Task<IActionResult> GetAll(CancellationToken cancellationToken)
    {
        var tasks = await taskService.GetAllAsync(currentUser.GetUserId(), cancellationToken);

        return Ok(ApiResponse<IReadOnlyList<TaskDto>>.Ok(tasks));
    }

    /// <summary>Получить задачу по идентификатору.</summary>
    [HttpGet("{id:long}")]
    [ProducesResponseType(typeof(ApiResponse<TaskDto>), StatusCodes.Status200OK)]
    [ProducesResponseType(typeof(ApiResponse<object?>), StatusCodes.Status401Unauthorized)]
    [ProducesResponseType(typeof(ApiResponse<object?>), StatusCodes.Status404NotFound)]
    public async Task<IActionResult> GetById(long id, CancellationToken cancellationToken)
    {
        var task = await taskService.GetByIdAsync(currentUser.GetUserId(), id, cancellationToken);

        return Ok(ApiResponse<TaskDto>.Ok(task));
    }

    /// <summary>Создать задачу.</summary>
    [HttpPost]
    [ProducesResponseType(typeof(ApiResponse<TaskDto>), StatusCodes.Status201Created)]
    [ProducesResponseType(typeof(ApiResponse<object?>), StatusCodes.Status400BadRequest)]
    [ProducesResponseType(typeof(ApiResponse<object?>), StatusCodes.Status401Unauthorized)]
    public async Task<IActionResult> Create(
        [FromBody] TaskInput input,
        CancellationToken cancellationToken)
    {
        var task = await taskService.CreateAsync(currentUser.GetUserId(), input, cancellationToken);

        return Created($"/api/tasks/{task.Id}", ApiResponse<TaskDto>.Ok(task, "Задача создана"));
    }
}
```

`[Authorize]` на контроллере закрывает **все** его методы: без токена придёт
`401` в нашей обёртке, ещё до входа в сервис.

# Если что-то не получилось

| Симптом | Что делать |
|---|---|
| `GetUserId()` бросает 401 при верном токене | claim назван иначе. В JWT это `sub`; проверьте `FindFirst("sub")` и `NameClaimType` в `TokenValidationParameters` |
| Задачи чужих пользователей видны | в предикате запроса нет `x.UserId == userId` — найдите такую выборку и исправьте |
| `POST /api/tasks` возвращает 400 про `completed_at` | `CompletedAt` не выставлен при `Status = Done`; см. страницу 09 |
| `429`/`401` на любом запросе | добавьте `builder.Services.AddHttpContextAccessor()` |
| `500` при `GetById` вместо 404 | `FirstOrDefaultAsync` вернул `null` и вы попытались обратиться к свойству без проверки |
