# 1. Экран сводки (110–130 мин)

`src/web/src/pages/SummaryPage.tsx`:

```tsx
import { useEffect, useState } from 'react';
import { errorMessage, tasksApi } from '../api/client';
import type { TaskSummary } from '../api/types';
import { Header } from '../components/Header';
import { Loader } from '../components/Loader';
import styles from './SummaryPage.module.css';

const PRIORITY_LABELS: Record<string, string> = {
  low: 'Низкий',
  medium: 'Средний',
  high: 'Высокий',
};

export function SummaryPage() {
  const [summary, setSummary] = useState<TaskSummary | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;

    tasksApi
      .summary()
      .then((data) => {
        if (!cancelled) setSummary(data);
      })
      .catch((e: unknown) => {
        if (!cancelled) setError(errorMessage(e));
      })
      .finally(() => {
        if (!cancelled) setLoading(false);
      });

    return () => {
      cancelled = true;
    };
  }, []);

  const cards = summary
    ? [
        { label: 'Всего задач', value: summary.total, tone: 'neutral' },
        { label: 'Выполнено', value: summary.done, tone: 'success' },
        { label: 'Не выполнено', value: summary.notDone, tone: 'neutral' },
        { label: 'Просрочено', value: summary.overdue, tone: summary.overdue > 0 ? 'danger' : 'neutral' },
      ]
    : [];

  const maxPriority = summary
    ? Math.max(1, ...Object.values(summary.byPriority))
    : 1;

  return (
    <div>
      <Header />

      <main className={styles.content}>
        <h1>Сводка</h1>

        {error && <div className="error" role="alert">{error}</div>}

        {loading && <Loader />}

        {!loading && summary && (
          <>
            <section className={styles.cards}>
              {cards.map((card) => (
                <div key={card.label} className={`${styles.card} ${styles[card.tone]}`}>
                  <span className={styles.value}>{card.value}</span>
                  <span className={styles.label}>{card.label}</span>
                </div>
              ))}
            </section>

            <section className={styles.panel}>
              <h2>Незавершённые задачи по приоритетам</h2>

              <ul className={styles.bars}>
                {(['low', 'medium', 'high'] as const).map((key) => {
                  const count = summary.byPriority[key] ?? 0;
                  const percent = Math.round((count / maxPriority) * 100);

                  return (
                    <li key={key} className={styles.barRow}>
                      <span className={styles.barLabel}>{PRIORITY_LABELS[key]}</span>
                      <div className={styles.barTrack}>
                        <div
                          className={`${styles.barFill} ${styles[key]}`}
                          style={{ width: `${percent}%` }}
                        />
                      </div>
                      <span className={styles.barValue}>{count}</span>
                    </li>
                  );
                })}
              </ul>

              <p className={styles.hint}>
                В сводку попадают только незавершённые задачи: выполненные уже не требуют внимания.
              </p>
            </section>
          </>
        )}
      </main>
    </div>
  );
}
```

# 2. Адаптивность 360–1920 px (130–140 мин)

Создайте `src/web/src/styles/responsive.css` и подключите его в
`main.tsx` после `global.css`:

```css
/* Правила, которые работают на любой странице */

.container {
  width: 100%;
  max-width: 1200px;
  margin: 0 auto;
  padding-left: 16px;
  padding-right: 16px;
}

/* До 900 px: двухколоночные формы становятся одноколоночными */
@media (max-width: 900px) {
  .row {
    grid-template-columns: 1fr;
  }
}

/* До 640 px: телефонная вёрстка */
@media (max-width: 640px) {
  html {
    font-size: 15px;
  }

  .content {
    padding-left: 12px;
    padding-right: 12px;
  }

  /* Горизонтальная прокрутка — главный признак сломанной адаптивности */
  .content,
  .form,
  .filters,
  .panel {
    min-width: 0;
  }

  img {
    max-width: 100%;
  }
}

/* Уважаем системную настройку «уменьшить движение» */
@media (prefers-reduced-motion: reduce) {
  * {
    animation-duration: 0.01ms !important;
    transition-duration: 0.01ms !important;
  }
}
```

# 3. Как проверять адаптивность (140–150 мин)

| Ширина | Что проверяем |
|---|---|
| 360 px | нижняя граница из задания; самое узкое место, ничего не обрезано |
| 768 px | планшет, две колонки |
| 1280 px | обычный ноутбук |
| 1920 px | верхняя граница из задания |

Проверка в браузере:

1. DevTools → `Ctrl`+`Shift`+`M` (Device Toolbar).
2. Выберите «Ответить на запросы» → введите `360 x 800`.
3. Пройдите `/tasks`, `/tasks/new`, `/summary`.
4. Ни на одной странице не должно быть горизонтальной полосы прокрутки.
5. Проверьте DevTools → Console: ошибок нет.
6. Повторите для `1920 x 1080`.

Быстрая проверка скриптом в консоли браузера:

```javascript
// ширина документа шире окна — значит есть горизонтальная прокрутка
document.documentElement.scrollWidth > window.innerWidth
```

```javascript
// найти элементы шире окна
[...document.querySelectorAll('*')]
  .filter((el) => el.getBoundingClientRect().width > window.innerWidth)
  .map((el) => `${el.tagName}.${el.className}`.slice(0, 60));
```

Два правила, которые чаще всего ломают адаптивность:

| Проблема | Причина | Исправление |
|---|---|---|
| горизонтальная прокрутка | `width: 1000px` или отсутствие `box-sizing: border-box` | `box-sizing` уже включён в `global.css`; фиксированные ширины заменить на `max-width` |
| текст вылезает из карточки | grid/flex-детей с `min-width: auto` | добавить `min-width: 0` контейнеру |
| таблица не влезает | таблица шире экрана | обернуть в контейнер `overflow-x: auto` |

# Проверка

1. Откройте `/summary` → четыре карточки с числами и три полосы приоритетов.
2. Числа совпадают с `GET /api/tasks/summary`:

```bash
curl -s http://localhost:5173/api/tasks/summary -H "Authorization: Bearer $TOKEN"
```

3. Удалите все задачи (или войдите новым пользователем без задач) →
   показываются нули, полосы приоритетов нулевые, страница не падает.
4. Остановите API → вместо пустого экрана сообщение «Нет связи с сервером».
5. Пройдите все страницы на 360 px, 768 px, 1280 px, 1920 px —
   горизонтальной прокрутки нет.
6. `document.documentElement.scrollWidth > window.innerWidth` даёт `false`
   на всех четырёх ширинах.
