Сессия 9. Тест-кейсы и автотесты · 90–120 мин

На региональном этапе 2026 за тестирование дают 4,5 %, и требуют **10
тест-кейсов** по шаблону плюс автоматизированные UI-тесты. На отборе это
остаётся лучшим способом показать, что вы умеете проверять свою работу.

# 2. Автотесты API (105–115 мин)

`src/api/TaskPlanner.Tests/TaskServiceTests.cs` — юнит-тесты сервиса на
InMemory-провайдере:

```csharp
using FluentAssertions;
using Microsoft.EntityFrameworkCore;
using TaskPlanner.Api.Data;
using TaskPlanner.Api.Services;
using TaskPlanner.Core.Common;
using TaskPlanner.Core.Dtos;
using TaskPlanner.Core.Enums;
using TaskPlanner.Core.Exceptions;
using Xunit;

namespace TaskPlanner.Tests;

public class TaskServiceTests : IDisposable
{
    private readonly AppDbContext _db;

    public TaskServiceTests()
    {
        // InMemory достаточно: проверяем логику сервиса, а не поведение
        // PostgreSQL. Ограничения БД проверяются в check_constraints.sql.
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseInMemoryDatabase(Guid.NewGuid().ToString())
            .Options;

        _db = new AppDbContext(options);
    }

    private TaskService CreateService()
        => new(_db, NullLogger<TaskService>.Instance);

    private long SeedUser(long userId)
    {
        _db.Users.Add(new Core.Entities.User
        {
            Id = userId,
            Email = $"user{userId}@college.ru",
            PasswordHash = "hash",
            FullName = $"Пользователь {userId}",
            CreatedAt = DateTimeOffset.UtcNow,
        });

        _db.SaveChanges();

        return userId;
    }

    private long SeedTask(long userId, string title = "Задача")
    {
        var task = new Core.Entities.TaskItem
        {
            UserId = userId,
            Title = title,
            Status = TaskStatus.New,
            Priority = TaskPriority.Medium,
            CreatedAt = DateTimeOffset.UtcNow,
            UpdatedAt = DateTimeOffset.UtcNow,
        };

        _db.Tasks.Add(task);
        _db.SaveChanges();

        return task.Id;
    }

    [Fact]
    public async Task CreateAsync_ОтсутствуетCompletedAt_УНевыполненной()
    {
        SeedUser(1);

        var created = await CreateService().CreateAsync(
            1,
            new TaskInput { Title = "Новая задача", Status = TaskStatus.New },
            CancellationToken.None);

        created.CompletedAt.Should().BeNull();
    }

    [Fact]
    public async Task CreateAsync_СтатусDone_ПроставляетCompletedAt()
    {
        SeedUser(1);

        var created = await CreateService().CreateAsync(
            1,
            new TaskInput { Title = "Сделана", Status = TaskStatus.Done },
            CancellationToken.None);

        created.Status.Should().Be(TaskStatus.Done);
        created.CompletedAt.Should().NotBeNull();
    }

    [Fact]
    public async Task GetByIdAsync_ЧужаяЗадача_ВозвращаетNotFound()
    {
        SeedUser(1);
        SeedUser(2);

        var taskId = SeedTask(userId: 1);

        var act = () => CreateService().GetByIdAsync(2, taskId, CancellationToken.None);

        // Не 403: существование чужой задачи не подтверждаем
        await act.Should().ThrowAsync<ApiException>()
            .Where(e => e.StatusCode == System.Net.HttpStatusCode.NotFound);
    }

    [Fact]
    public async Task GetAllAsync_ВозвращаетТолькоСвоиЗадачи()
    {
        SeedUser(1);
        SeedUser(2);

        SeedTask(1, "Задача первого");
        SeedTask(2, "Задача второго");

        var tasks = await CreateService().GetAllAsync(1, CancellationToken.None);

        tasks.Should().HaveCount(1);
        tasks[0].Title.Should().Be("Задача первого");
    }

    [Fact]
    public async Task ToggleCompletedAsync_Дважды_ВозвращаетВНовую()
    {
        SeedUser(1);
        var taskId = SeedTask(1);

        var service = CreateService();

        var done = await service.ToggleCompletedAsync(1, taskId, CancellationToken.None);
        done.Status.Should().Be(TaskStatus.Done);
        done.CompletedAt.Should().NotBeNull();

        var back = await service.ToggleCompletedAsync(1, taskId, CancellationToken.None);
        back.Status.Should().Be(TaskStatus.New);
        back.CompletedAt.Should().BeNull();
    }

    [Fact]
    public async Task DeleteAsync_ЧужаяЗадача_ВозвращаетNotFound()
    {
        SeedUser(1);
        SeedUser(2);
        var taskId = SeedTask(1);

        var act = () => CreateService().DeleteAsync(2, taskId, CancellationToken.None);

        await act.Should().ThrowAsync<ApiException>();

        _db.Tasks.Count().Should().Be(1);
    }

    public void Dispose() => _db.Dispose();
}
```

`src/web/e2e/tasks.spec.ts`:

```typescript
import { test, expect } from '@playwright/test';

const API = 'http://localhost:5000';

test.describe('Список задач', () => {
  let token = '';
  let email = `e2e-${Date.now()}@college.ru`;

  test.beforeAll(async () => {
    const register = await fetch(`${API}/api/auth/register`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password: 'Passw0rd123', fullName: 'E2E Пользователь' }),
    });

    expect(register.status).toBe(201);

    const login = await fetch(`${API}/api/auth/login`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password: 'Passw0rd123' }),
    });

    token = (await login.json()).data.token;

    for (const title of ['Первая задача E2E', 'Вторая задача E2E']) {
      await fetch(`${API}/api/tasks`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
        body: JSON.stringify({ title }),
      });
    }
  });

  test('неавторизованный пользователь попадает на вход', async ({ page }) => {
    await page.goto('/tasks');

    await expect(page).toHaveURL(/\/login/);
  });

  test('после входа открывается список', async ({ page }) => {
    await page.goto('/login');
    await page.getByLabel('Email').fill(email);
    await page.getByLabel('Пароль').fill('Passw0rd123');
    await page.getByRole('button', { name: 'Войти' }).click();

    await expect(page).toHaveURL(/\/tasks/);
    await expect(page.getByText('Первая задача E2E')).toBeVisible();
  });

  test('неверный пароль показывает сообщение', async ({ page }) => {
    await page.goto('/login');
    await page.getByLabel('Email').fill(email);
    await page.getByLabel('Пароль').fill('WrongPassw0rd');
    await page.getByRole('button', { name: 'Войти' }).click();

    await expect(page.getByRole('alert')).toContainText('Неверный email или пароль');
  });

  test('поиск без совпадений показывает пустое состояние', async ({ page }) => {
    await page.goto('/tasks');

    await page.evaluate(() => {
      localStorage.setItem('taskplanner.token', '');
    });
    await page.goto('/login');
    await page.getByLabel('Email').fill(email);
    await page.getByLabel('Пароль').fill('Passw0rd123');
    await page.getByRole('button', { name: 'Войти' }).click();
    await expect(page.getByText('Первая задача E2E')).toBeVisible();

    await page.getByPlaceholder('по названию или описанию').fill('щщщщ');
    await page.getByRole('button', { name: 'Применить' }).click();

    await expect(page.getByText('Ничего не найдено')).toBeVisible();
  });
});
```

# Если что-то не получилось

| Симптом | Что делать |
|---|---|
| `UseInMemoryDatabase` не поддерживает `HasColumnType` | InMemory игнорирует типы колонок — это нормально |
| `NullLogger` не найден | `using Microsoft.Extensions.Logging.Abstractions;` |
| Тесты на InMemory не ловят нарушения CHECK | для этого есть `db/check_constraints.sql` — ограничения проверяются на реальном PostgreSQL |
| `dotnet test` не находит проекты | запускайте из `src`, где лежит `.sln` |
| Playwright: `npx playwright install` требует sudo | на РедОС используйте `--with-deps` и `sudo` один раз |
| Playwright: тесты падают из-за гонки | включите `reuseExistingServer: true` и поднимите API заранее |
| `getByLabel` не находит поле | в разметке нет `<label for>…</label>` |
