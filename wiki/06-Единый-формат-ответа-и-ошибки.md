Сессия 2. Backend: единый формат ответа · 75–150 мин

# 1. Обёртка ответа (75–90 мин)

По [`criteria.md`](../criteria.md) за единый формат ответа на всех методах дают
баллы. Формат зафиксирован в [`api/openapi.yaml`](../api/openapi.yaml):

```json
{ "success": true, "data": { }, "message": "OK", "error_code": null }
{ "success": false, "data": null, "message": "Задача не найдена", "error_code": "NOT_FOUND" }
```

Обратите внимание: `data` есть **всегда** (может быть `null`), а `error_code`
записан в `snake_case`.

`src/api/TaskPlanner.Core/Common/ApiResponse.cs`:

```csharp
using System.Text.Json.Serialization;

namespace TaskPlanner.Core.Common;

/// <summary>
/// Единая обёртка ответа всех методов API.
/// Клиенты (web, desktop, mobile) разбирают только её.
/// </summary>
public class ApiResponse<T>
{
    [JsonPropertyName("success")]
    public bool Success { get; init; }

    [JsonPropertyName("data")]
    public T? Data { get; init; }

    [JsonPropertyName("message")]
    public string Message { get; init; } = string.Empty;

    [JsonPropertyName("error_code")]
    public string? ErrorCode { get; init; }

    public static ApiResponse<T> Ok(T? data, string message = "OK")
        => new() { Success = true, Data = data, Message = message, ErrorCode = null };

    public static ApiResponse<T> Fail(string message, string errorCode)
        => new() { Success = false, Data = default, Message = message, ErrorCode = errorCode };
}
```

`src/api/TaskPlanner.Core/Common/ErrorCodes.cs`:

```csharp
namespace TaskPlanner.Core.Common;

/// <summary>
/// Коды ошибок из контракта API. Клиенты ориентируются на них,
/// а не на текст message.
/// </summary>
public static class ErrorCodes
{
    public const string ValidationError = "VALIDATION_ERROR";
    public const string Unauthorized = "UNAUTHORIZED";
    public const string Forbidden = "FORBIDDEN";
    public const string NotFound = "NOT_FOUND";
    public const string InternalError = "INTERNAL_ERROR";
    public const string EmailTaken = "EMAIL_TAKEN";
}
```

> **Замечание:** если `data` — это `void` (например, `DELETE` с кодом 204),
  используйте `ApiResponse<object?>`. Нетипизированный `object` в обёртке
  заставляет клиента приводить тип вручную; типизированная обёртка даёт
  проверку на этапе компиляции.

# 2. Исключение предметной области (90–100 мин)

`src/api/TaskPlanner.Core/Exceptions/ApiException.cs`:

```csharp
using System.Net;

namespace TaskPlanner.Core.Exceptions;

/// <summary>
/// Ожидаемая ошибка бизнес-логики. Middleware превращает её в ответ API.
/// Исключения, которые вы не ожидали, — это уже 500 и баг.
/// </summary>
public class ApiException(
    HttpStatusCode statusCode,
    string errorCode,
    string message) : Exception(message)
{
    public HttpStatusCode StatusCode { get; } = statusCode;

    public string ErrorCode { get; } = errorCode;

    public static ApiException NotFound(string message)
        => new(HttpStatusCode.NotFound, Common.ErrorCodes.NotFound, message);

    public static ApiException BadRequest(string message)
        => new(HttpStatusCode.BadRequest, Common.ErrorCodes.ValidationError, message);

    public static ApiException Unauthorized(string message)
        => new(HttpStatusCode.Unauthorized, Common.ErrorCodes.Unauthorized, message);

    public static ApiException Forbidden(string message)
        => new(HttpStatusCode.Forbidden, Common.ErrorCodes.Forbidden, message);

    public static ApiException Conflict(string errorCode, string message)
        => new(HttpStatusCode.Conflict, errorCode, message);
}
```

# 3. Middleware ошибок (100–125 мин)

`src/api/TaskPlanner.Api/Infrastructure/ApiExceptionMiddleware.cs`:

```csharp
using System.Net;
using System.Text.Json;
using FluentValidation;
using Microsoft.EntityFrameworkCore;
using Npgsql;
using TaskPlanner.Core.Common;
using TaskPlanner.Core.Exceptions;

namespace TaskPlanner.Api.Infrastructure;

/// <summary>
/// Единая точка, где исключение превращается в ответ API.
/// Регистрируется первым, чтобы ловить ошибки всех остальных middleware.
/// </summary>
public class ApiExceptionMiddleware(
    RequestDelegate next,
    ILogger<ApiExceptionMiddleware> logger,
    IHostEnvironment environment)
{
    private static readonly JsonSerializerOptions JsonOptions = new(JsonSerializerDefaults.Web);

    public async Task InvokeAsync(HttpContext context)
    {
        try
        {
            await next(context);
        }
        catch (Exception exception)
        {
            await WriteErrorAsync(context, exception);
        }
    }

    private async Task WriteErrorAsync(HttpContext context, Exception exception)
    {
        HttpStatusCode statusCode;
        string errorCode;
        string message;

        switch (exception)
        {
            case ApiException api:
                statusCode = api.StatusCode;
                errorCode = api.ErrorCode;
                message = api.Message;
                logger.LogInformation(
                    "Ожидаемая ошибка {ErrorCode}: {Message}", errorCode, api.Message);
                break;

            case ValidationException validation:
                statusCode = HttpStatusCode.BadRequest;
                errorCode = ErrorCodes.ValidationError;
                message = string.Join("; ", validation.Errors.Select(x => x.ErrorMessage));
                logger.LogInformation("Ошибка валидации: {Message}", message);
                break;

            case KeyNotFoundException notFound:
                statusCode = HttpStatusCode.NotFound;
                errorCode = ErrorCodes.NotFound;
                message = notFound.Message;
                break;

            case UnauthorizedAccessException unauthorized:
                statusCode = HttpStatusCode.Unauthorized;
                errorCode = ErrorCodes.Unauthorized;
                message = unauthorized.Message;
                break;

            case DbUpdateException dbUpdate:
                // Нарушение уникальности или CHECK-ограничения в базе.
                // Текст ошибки PostgreSQL не отдаём наружу: он раскрывает
                // структуру таблиц и имена ограничений.
                statusCode = HttpStatusCode.BadRequest;
                errorCode = ErrorCodes.ValidationError;
                message = "Запись не может быть сохранена: нарушено ограничение данных";
                logger.LogWarning(
                    dbUpdate, "Ошибка сохранения в базе данных");
                break;

            default:
                statusCode = HttpStatusCode.InternalServerError;
                errorCode = ErrorCodes.InternalError;
                message = "Внутренняя ошибка сервера";
                logger.LogError(exception, "Необработанное исключение");
                break;
        }

        var traceId = context.TraceIdentifier;

        var response = new ApiResponse<object?>
        {
            Success = false,
            Data = null,
            Message = $"{message} (traceId: {traceId})",
            ErrorCode = errorCode,
        };

        if (context.Response.HasStarted)
        {
            logger.LogWarning("Ответ уже начат отправлять, тело ошибки не изменить");
            return;
        }

        context.Response.Clear();
        context.Response.StatusCode = (int)statusCode;
        context.Response.ContentType = "application/json; charset=utf-8";

        await context.Response.WriteAsync(
            JsonSerializer.Serialize(response, JsonOptions));
    }
}
```

Проверка конкретной ошибки PostgreSQL — если нужно различить дубликат email и
остальные нарушения, посмотрите на тип исключения:

```csharp
case DbUpdateException dbUpdate
    when dbUpdate.InnerException is PostgresException { SqlState: PostgresErrorCodes.UniqueViolation }:
    statusCode = HttpStatusCode.Conflict;
    errorCode = ErrorCodes.EmailTaken;
    message = "Пользователь с таким email уже существует";
    logger.LogInformation("Нарушение уникальности: {Message}", dbUpdate.InnerException.Message);
    break;
```

> **Замечание:** положите этот `case` **перед** общим `DbUpdateException` —
> в C# первое подходящее выигрывает, порядок важен. Нарушение уникальности и
> нарушение `CHECK` не должны выглядеть одинаково.

# 4. `Program.cs` (125–145 мин)

Замените содержимое `src/api/TaskPlanner.Api/Program.cs`:

```csharp
using System.Text.Json;
using System.Text.Json.Serialization;
using FluentValidation;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using TaskPlanner.Api.Data;
using TaskPlanner.Api.Infrastructure;
using TaskPlanner.Core.Common;
using TaskPlanner.Core.Enums;

var builder = WebApplication.CreateBuilder(args);

// ---------------------------------------------------------------------
// База данных
// ---------------------------------------------------------------------
builder.Services.AddDbContext<AppDbContext>(options =>
    options.UseNpgsql(
        builder.Configuration.GetConnectionString("DefaultConnection"),
        npgsql => npgsql.MigrationsAssembly(typeof(AppDbContext).Assembly.FullName)));

// ---------------------------------------------------------------------
// JSON: контракт требует in_progress, а не inProgress
// ---------------------------------------------------------------------
builder.Services
    .AddControllers()
    .AddJsonOptions(options =>
    {
        options.JsonSerializerOptions.PropertyNamingPolicy = JsonNamingPolicy.CamelCase;
        options.JsonSerializerOptions.Converters.Add(new TaskStatusConverter());
        options.JsonSerializerOptions.Converters.Add(new TaskPriorityConverter());
        options.JsonSerializerOptions.DefaultIgnoreCondition =
            JsonIgnoreCondition.WhenWritingNull;
    });

// Включаем camelCase и в Swagger, иначе он покажет PascalCase
builder.Services.Configure<Microsoft.AspNetCore.Mvc.ApiBehaviorOptions>(options =>
{
    options.SuppressModelStateInvalidFilter = true;
});

builder.Services.AddEndpointsApiExplorer();

// FluentValidation: валидаторы из Core подхватываются автоматически
builder.Services.AddValidatorsFromAssemblyContaining<TaskStatusConverter>();

// ---------------------------------------------------------------------
// CORS: web-клиент живёт на другом порту
// ---------------------------------------------------------------------
const string CorsPolicy = "web-client";
var allowedOrigins = builder.Configuration
    .GetSection("Cors:AllowedOrigins")
    .Get<string[]>() ?? ["http://localhost:5173"];

builder.Services.AddCors(options =>
{
    options.AddPolicy(CorsPolicy, policy => policy
        .WithOrigins(allowedOrigins)
        .AllowAnyHeader()
        .AllowAnyMethod());
});

// ---------------------------------------------------------------------
// Swagger с кнопкой Authorize (JWT)
// ---------------------------------------------------------------------
builder.Services.AddSwaggerGen(options =>
{
    options.SwaggerDoc("v1", new Microsoft.OpenApi.Models.OpenApiInfo
    {
        Title = "Планировщик личных задач — API",
        Version = "v1",
        Description = "Общий REST API для web-, desktop- и mobile-клиентов",
    });

    var scheme = new Microsoft.OpenApi.Models.OpenApiSecurityScheme
    {
        Name = "Authorization",
        Type = Microsoft.OpenApi.Models.SecuritySchemeType.Http,
        Scheme = "bearer",
        BearerFormat = "JWT",
        In = Microsoft.OpenApi.Models.ParameterLocation.Header,
        Description = "Вставьте JWT-токен, полученный в /api/auth/login",
        Reference = new Microsoft.OpenApi.Models.OpenApiReference
        {
            Type = Microsoft.OpenApi.Models.ReferenceType.SecurityScheme,
            Id = "Bearer",
        },
    };

    options.AddSecurityDefinition("Bearer", scheme);
    options.AddSecurityRequirement(
        new Microsoft.OpenApi.Models.OpenApiSecurityRequirement
        {
            [scheme] = Array.Empty<string>(),
        });
});

var app = builder.Build();

// ---------------------------------------------------------------------
// Порядок middleware важен
// ---------------------------------------------------------------------
// 1. Ошибки — ловят всё, что ниже
app.UseMiddleware<ApiExceptionMiddleware>();

// 2. CORS — до аутентификации, иначе preflight-запросы блокируются
app.UseCors(CorsPolicy);

app.UseSwagger();
app.UseSwaggerUI(options =>
{
    options.SwaggerEndpoint("/swagger/v1/swagger.json", "TaskPlanner API v1");
    options.RoutePrefix = "swagger";
});

// 3. Маршрутизация контроллеров
app.MapControllers();

app.Run();
```

# 5. `GET /api/health` (145–150 мин)

`src/api/TaskPlanner.Api/Controllers/HealthController.cs`:

```csharp
using Microsoft.AspNetCore.Mvc;
using TaskPlanner.Api.Data;
using TaskPlanner.Core.Common;

namespace TaskPlanner.Api.Controllers;

[ApiController]
[Route("api/health")]
public class HealthController(AppDbContext db) : ControllerBase
{
    /// <summary>Проверка живости сервиса и связи с базой данных.</summary>
    [HttpGet]
    public async Task<IActionResult> Get(CancellationToken cancellationToken)
    {
        var canConnect = await db.Database.CanConnectAsync(cancellationToken);

        var response = ApiResponse<object?>.Ok(
            new
            {
                status = canConnect ? "ok" : "degraded",
                database = canConnect ? "connected" : "unavailable",
                utc = DateTime.UtcNow,
            },
            canConnect ? "OK" : "Сервер работает, база данных недоступна");

        return canConnect ? Ok(response) : StatusCode(
            StatusCodes.Status503ServiceUnavailable, response);
    }
}
```

# Если что-то не получилось

| Симптом | Что делать |
|---|---|
| `ApiResponse<T>` не сериализуется, имена PascalCase | не забудьте `[JsonPropertyName]` или проверьте `AddControllers().AddJsonOptions` |
| В Swagger вместо `in_progress` — `InProgress` | конвертеры не зарегистрированы в `AddJsonOptions`; для Swagger добавьте `SchemaFilter` |
| `ValidationException` не ловится | не хватает `using FluentValidation;` в middleware |
| Middleware не срабатывает | `app.UseMiddleware<ApiExceptionMiddleware>()` должен быть **первым** вызовом |
| `error_code` вместо `errorCode` в JSON | проверьте `[JsonPropertyName("error_code")]` |
| Клиент жалуется на CORS | адрес web-клиента должен быть ровно `http://localhost:5173` и быть в `Cors:AllowedOrigins` |
